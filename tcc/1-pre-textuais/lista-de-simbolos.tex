\begin{simbolos} \itemsep -1pt
    \item[$V_t$] Volume armazenado no início do mês $t$, em hm$^3$
    \item[$V_{t+1}$] Volume armazenado ao final do mês $t$, em hm$^3$
    \item[$I_t$] Volume afluente no mês $t$, em hm$^3$
    \item[$T_t$] Saldo de transferências no mês $t$, em hm$^3$
    \item[$R_t$] Retirada efetiva no mês $t$, em hm$^3$
    \item[$E_t$] Volume evaporado no mês $t$, em hm$^3$
    \item[$S_t$] Volume vertido no mês $t$, em hm$^3$
    \item[$e_t$] Lâmina de evaporação no mês $t$, em m
    \item[$A$] Área da superfície do reservatório, em km$^2$
    \item[$Q_t$] Vazão média afluente no mês $t$, em m$^3$/s
    \item[$P_t$] Volume preliminar do mês $t$, antes dos limites físicos, em hm$^3$
    \item[$D_t$] Demanda solicitada no mês $t$
    \item[$D^{a}_t$] Demanda aplicada no mês $t$, após o racionamento
    \item[$V^{*}_{t+1}$] Primeira estimativa do volume final, em hm$^3$
    \item[$A_t$] Área da superfície no início do mês $t$, em km$^2$
    \item[$A^{*}_{t+1}$] Área correspondente à primeira estimativa do volume final, em km$^2$
    \item[$V_{\max}$] Capacidade do reservatório, em hm$^3$
    \item[$\Delta_t$] Déficit de atendimento no mês $t$, em hm$^3$
    \item[$p_t$] Volume inicial do mês $t$ em percentual da capacidade
    \item[$r_k$] Percentual de racionamento do estado $k$
    \item[$k$] Índice do estado operacional
    \item[$\varepsilon_t$] Resíduo de conservação de massa no mês $t$, em hm$^3$
\end{simbolos}
